% example environment to print a gray bar next to the text. This is very useful to highlight 
% some text passages

\renewenvironment{leftbar}[1][\hsize]
{%
    \def\FrameCommand
    {%
        
        {\color{gray}\vrule width 5pt}%
        %{\color{green}\rule[0pt]{5pt}{\vsize}}%
        \hspace{0pt}%must no space.
        \fboxsep=\FrameSep\colorbox{white}%
    }%
    \MakeFramed{\hsize#1\advance\hsize-\width\FrameRestore}%
}
{\endMakeFramed}
